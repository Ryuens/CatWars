## 계산 공식

### 최종 효과 계산
```
최종 효과 = BaseValue × Multiplier
예: 전설 공격력 Lv.50 = 12.0 × 3.45 = 41.4%
```

### 다음 레벨 필요 개수
```
필요 개수 = 현재 레벨
예: Lv.25 → Lv.26 = 25개 필요
```

### 만렙까지 총 필요 개수
```
총 개수 = 100 × 99 / 2 = 4,950개
```

## 구현 참고 코드

### 장비 효과 계산
```csharp
// 장비 최종 효과 계산
float GetEquipmentEffect(string equipmentIdx, int level)
{
    var equipment = GetEquipment(equipmentIdx);
    var levelData = GetEquipmentLevel(equipmentIdx, level);

    return equipment.BaseValue * levelData.Multiplier;
}

// 예시: E09 용암대검 Lv.50
// BaseValue = 12.0, Multiplier = 3.45
// 결과: 12.0 * 3.45 = 41.4%
```

### 자동 강화 (중복 획득)
```csharp
// 장비 획득 시 자동 강화
void OnEquipmentAcquired(string equipmentIdx)
{
    var playerEquipment = GetPlayerEquipment(equipmentIdx);

    if (playerEquipment == null)
    {
        // 최초 획득
        CreatePlayerEquipment(equipmentIdx, 1);
    }
    else
    {
        // 중복 획득 - 자동 강화
        int currentLevel = playerEquipment.Level;
        int requiredCount = currentLevel; // 다음 레벨 필요 개수

        playerEquipment.DuplicateCount++;

        if (playerEquipment.DuplicateCount >= requiredCount)
        {
            // 레벨업
            playerEquipment.Level++;
            playerEquipment.DuplicateCount = 0;
        }
    }
}
```

### 유닛 장비 장착
```csharp
// 유닛당 1개 장비 장착
bool EquipToUnit(int unitIdx, string equipmentIdx)
{
    var unit = GetUnit(unitIdx);
    var equipment = GetPlayerEquipment(equipmentIdx);

    if (equipment == null)
        return false;

    // 기존 장비가 있다면 해제
    if (unit.EquippedItem != null)
    {
        UnequipFromUnit(unitIdx);
    }

    // 새 장비 장착
    unit.EquippedItem = equipmentIdx;
    unit.EquipmentLevel = equipment.Level;

    // 스탯 적용
    ApplyEquipmentStats(unitIdx);

    return true;
}

// 스탯 적용
void ApplyEquipmentStats(int unitIdx)
{
    var unit = GetUnit(unitIdx);

    if (unit.EquippedItem == null)
        return;

    float effect = GetEquipmentEffect(unit.EquippedItem, unit.EquipmentLevel);
    var equipment = GetEquipment(unit.EquippedItem);

    switch (equipment.EffectType)
    {
        case 1: // 공격력
            unit.BonusAttack += effect;
            break;
        case 2: // 체력
            unit.BonusHealth += effect;
            break;
        case 3: // 공격속도
            unit.BonusAttackSpeed += effect;
            break;
        case 4: // 치명타
            unit.BonusCritRate += effect;
            break;
        case 5: // 모든 스탯
            unit.BonusAttack += effect;
            unit.BonusHealth += effect;
            unit.BonusAttackSpeed += effect;
            unit.BonusCritRate += effect;
            break;
    }
}
```

    foreach (var pity in pityData)
    {
        if (playerData.SummonCount >= pity.RequiredCount)
        {
            // 확정 등급 보장
            GiveGuaranteedEquipment(pity.GuaranteedGrade);
            playerData.SummonCount = 0;
            return;
        }
    }

    // 일반 소환
    GiveRandomEquipment();
}
