# 유닛 장비 데이터 테이블

## 개요
유닛 장비 시스템 구현을 위한 데이터 테이블 스펙 및 전체 데이터

---

## 테이블 목록

| 테이블명 | 설명 |
|----------|------|
| Equipment.xlsx | 장비 기본 정보 |
| EquipmentStat.xlsx | 장비 스탯 상세 |
| EquipmentEnhance.xlsx | 강화 비용/확률 |
| EquipmentSummon.xlsx | 소환 확률/비용 |
| EquipmentSet.xlsx | 세트 효과 |
| EquipmentSlot.xlsx | 슬롯 해금 조건 |
| EnhanceMaterial.xlsx | 강화 재료 |

## Equipment.xlsx (장비 기본 정보)

### 테이블 구조
| 컬럼명 | 타입 | 설명 |
|--------|------|------|
| EquipmentIdx | int | 장비 고유 ID |
| Name | string | 장비 이름 |
| SlotType | int | 슬롯 타입 (1:무기, 2:방어구, 3:투구, 4:악세) |
| SubType | int | 세부 타입 |
| Grade | int | 등급 (1:일반~5:신화) |
| SetIdx | int | 세트 ID (0=세트 없음) |
| UnitType | int | 장착 가능 유닛 타입 (0:전체, 1:근거리, 2:원거리) |
| IconPath | string | 아이콘 경로 |
| Description | string | 설명 |

## 구현 참고사항

### 장비 장착 로직
```csharp
// 장착 가능 여부 확인
bool CanEquip(int equipmentIdx, int unitIdx)
{
    var equipment = GetEquipment(equipmentIdx);
    var unit = GetUnit(unitIdx);

    // 슬롯 타입 확인
    if (!IsSlotUnlocked(equipment.SlotType))
        return false;

    // 유닛 타입 확인 (0=전체, 1=근거리, 2=원거리)
    if (equipment.UnitType != 0 && equipment.UnitType != unit.Type)
        return false;

    return true;
}
```

### 강화 로직
```csharp
// 강화 시도
bool TryEnhance(int equipmentIdx, int targetLevel)
{
    var enhance = GetEnhanceData(targetLevel);

    // 비용 확인
    if (Gold < enhance.GoldCost)
        return false;
    if (GetMaterialCount(enhance.MaterialGrade) < enhance.MaterialCount)
        return false;

    // 비용 차감
    Gold -= enhance.GoldCost;
    UseMaterial(enhance.MaterialGrade, enhance.MaterialCount);

    // 성공 확률 계산
    float rate = enhance.SuccessRate;
    if (HasBlessingScroll())
        rate += 10;

    // 결과 판정
    if (Random.Range(0, 100) < rate)
    {
        // 성공
        SetEquipmentLevel(equipmentIdx, targetLevel);
        return true;
    }
    else
    {
        // 실패 (레벨 유지)
        return false;
    }
}
```

### 세트 효과 계산
```csharp
// 세트 효과 계산
Dictionary<int, int> GetSetPieceCounts(int unitIdx)
{
    var result = new Dictionary<int, int>();
    var equipments = GetUnitEquipments(unitIdx);

    foreach (var eq in equipments)
    {
        if (eq.SetIdx > 0)
        {
            if (!result.ContainsKey(eq.SetIdx))
                result[eq.SetIdx] = 0;
            result[eq.SetIdx]++;
        }
    }

    return result;
}

// 세트 보너스 적용
void ApplySetBonus(int unitIdx)
{
    var setCounts = GetSetPieceCounts(unitIdx);

    foreach (var kvp in setCounts)
    {
        var setEffects = GetSetEffects(kvp.Key);
        foreach (var effect in setEffects)
        {
            if (kvp.Value >= effect.PieceCount)
            {
                ApplyStatBonus(unitIdx, effect.StatType, effect.StatValue);
            }
        }
    }
}
```
