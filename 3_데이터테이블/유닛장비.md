| RequiredDuplicates | int | 다음 레벨 필요 중복 개수 |
| RequiredGold | int | 다음 레벨 필요 골드 |
| TotalDuplicates | int | 해당 레벨까지 누적 중복 개수 |
| TotalGold | int | 해당 레벨까지 누적 골드 |

### 레벨 계산 공식

```
레벨 배율 = 1.0 + (Level - 1) × 0.05
다음 레벨 필요 중복 = Level
다음 레벨 필요 골드 = Level × 1,000
누적 중복 개수 = Level × (Level - 1) / 2
누적 골드 = Level × (Level - 1) / 2 × 1,000
```

## 계산 공식

### 최종 효과 계산
```
최종 효과 = BaseValue × LevelMultiplier × StarMultiplier

예시: 전설 공격력 유물 (R05 창조의검)
- 기본 효과: 30%
- Lv.100: 6.0x
- 5성: 2.5x
- 최종 효과: 30% × 6.0 × 2.5 = 450%
```

### 레벨업 필요 재료
```
중복 개수 = 현재 레벨
골드 = 현재 레벨 × 1,000

예: Lv.25 → Lv.26
- 중복 25개
- 골드 25,000
```

### 만렙까지 총 필요 재료
```
총 중복 개수 = 100 × 99 / 2 = 4,950개
총 골드 = 4,950 × 1,000 = 4,950,000
```

### 별 진화 비용
```
3★ → 4★: 상급석 15개 = 마법석 750개
4★ → 5★: 상급석 25개 = 마법석 1,250개
합계: 상급석 40개 = 마법석 2,000개
```

### 최종 완성 필요 재료
```
하나의 유물을 Lv.100 5★로 완성하려면:
- 중복 유물: 4,950개
- 골드: 4,950,000
- 마법석: 2,000개 (상급석 40개)
```

## 구현 참고 코드

### 유물 최종 효과 계산
```csharp
// 유물 최종 효과 계산
float GetRelicFinalEffect(string relicIdx, int level, int star, int effectIndex)
{
    var relic = GetRelic(relicIdx);
    var levelData = GetRelicLevel(relicIdx, level);
    var starData = GetRelicStar(star);

    float baseValue = (effectIndex == 1) ? relic.BaseValue1 : relic.BaseValue2;

    return baseValue * levelData.LevelMultiplier * starData.StarMultiplier;
}

// 예시: R05 창조의검 Lv.100 5★
// BaseValue1 = 30.0, LevelMultiplier = 6.0, StarMultiplier = 2.5
// 결과: 30.0 × 6.0 × 2.5 = 450%
```

### 유물 레벨업 (수동)
```csharp
// 유물 레벨업
bool LevelUpRelic(string relicIdx)
{
    var playerRelic = GetPlayerRelic(relicIdx);
    var levelData = GetRelicLevel(relicIdx, playerRelic.Level);

    if (playerRelic.Level >= 100)
        return false;

    // 중복 개수 확인
    if (playerRelic.DuplicateCount < levelData.RequiredDuplicates)
        return false;

    // 골드 확인
    if (playerData.Gold < levelData.RequiredGold)
        return false;

    // 레벨업
    playerRelic.Level++;
    playerRelic.DuplicateCount -= levelData.RequiredDuplicates;
    playerData.Gold -= levelData.RequiredGold;

    // 자동 별 진화 체크 (25레벨, 50레벨)
    if (playerRelic.Level == 25 && playerRelic.Star == 1)
        playerRelic.Star = 2;
    else if (playerRelic.Level == 50 && playerRelic.Star == 2)
        playerRelic.Star = 3;

    return true;
}
```

### 유물 별 진화 (수동)
```csharp
// 유물 별 진화
bool EvolveRelicStar(string relicIdx)
{
    var playerRelic = GetPlayerRelic(relicIdx);
    var evolutionData = GetRelicEvolution(playerRelic.Star);

    if (evolutionData == null)
        return false; // 더 이상 진화 불가

    // 레벨 체크
    if (playerRelic.Level < evolutionData.RequiredLevel)
        return false;

    // 상급석 체크
    if (playerData.SuperiorStone < evolutionData.RequiredSuperiorStone)
        return false;

    // 진화
    playerRelic.Star++;
    playerData.SuperiorStone -= evolutionData.RequiredSuperiorStone;

    return true;
}
```

// 유닛당 1개 유물 장착
bool EquipRelicToUnit(int unitIdx, string relicIdx)
{
    var unit = GetUnit(unitIdx);
    var relic = GetPlayerRelic(relicIdx);

    if (relic == null)
        return false;

    // 기존 유물이 있다면 해제
    if (unit.EquippedRelic != null)
    {
        UnequipRelicFromUnit(unitIdx);
    }

    // 새 유물 장착
    unit.EquippedRelic = relicIdx;
    unit.RelicLevel = relic.Level;
    unit.RelicStar = relic.Star;

    // 스탯 적용
    ApplyRelicStats(unitIdx);

    return true;
}

// 스탯 적용
void ApplyRelicStats(int unitIdx)
{
    var unit = GetUnit(unitIdx);

    if (unit.EquippedRelic == null)
        return;

    var relic = GetRelic(unit.EquippedRelic);

    // 효과 1 적용
    float effect1 = GetRelicFinalEffect(unit.EquippedRelic, unit.RelicLevel, unit.RelicStar, 1);
    ApplyStatBonus(unit, relic.EffectType1, effect1);

    // 효과 2 적용
    float effect2 = GetRelicFinalEffect(unit.EquippedRelic, unit.RelicLevel, unit.RelicStar, 2);
    ApplyStatBonus(unit, relic.EffectType2, effect2);
}

void ApplyStatBonus(Unit unit, int effectType, float value)
{
    switch (effectType)
    {
        case 1: // 공격력
            unit.BonusAttack += value;
            break;
        case 2: // 체력
            unit.BonusHealth += value;
            break;
        case 3: // 공격속도
            unit.BonusAttackSpeed += value;
            break;
        case 4: // 치명타
            unit.BonusCritRate += value;
            break;
        case 5: // 골드 획득
            unit.BonusGoldGain += value;
            break;
        case 6: // 소환석 생산
            unit.BonusSummonStone += value;
            break;
        case 7: // 스킬 쿨타임
            unit.BonusCooldownReduction += value;
            break;
        case 8: // 보스 피해
            unit.BonusBossDamage += value;
            break;
        case 9: // 요새 체력
            unit.BonusFortressHealth += value;
            break;
        case 10: // 모든 스탯
            unit.BonusAllStats += value;
            break;
    }
}
